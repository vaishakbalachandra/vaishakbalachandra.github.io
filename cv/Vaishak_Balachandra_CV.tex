% ================= THESIS =================
\cvsection{Master's Thesis}
\begin{cvindent}

\cvlab{Title.}{\emph{VertiSplitRAG: Privacy-Preserving Retrieval-Augmented Generation Through Vertical
Attribute Partitioning.} Purdue University Northwest, May 2026. \href{https://doi.org/10.25394/PGS.32166906}{doi:10.25394/PGS.32166906}}

\cvlab{Advisor.}{Prof.\ Shafkat Islam.\quad \textbf{Committee.} Prof.\ Hairong Zhao, Prof.\ Yang Ni.}

\cvlab{Summary.}{Retrieval-augmented generation systems embed every sensitive attribute of a record
jointly, and existing privacy methods apply protection only after the embeddings are formed, leaving
multi-attribute leakage unaddressed. This thesis treats the exposure as architectural rather than
procedural: what each component of the system is permitted to observe is a design decision. It proposes
a privacy-by-design framework that restructures data through vertical attribute partitioning
\emph{before} embedding, so that no single component ever observes a complete record. Four
architectural variants were implemented and evaluated on synthetic medical patient records --- Baseline
RAG (BAS), VertiSplitRAG for single-attribute queries (VSS), VertiSplitRAG for multi-attribute queries
with a shared LLM (VSM), and VertiSplitRAG with a multi-LLM architecture (VSML) --- measured on privacy
protection, answer quality and system latency. VSML achieves complete cross-attribute isolation through
\emph{information-theoretic structural separation}, maintaining the full privacy budget across all
queries where the other systems progressively exhaust their isolation guarantees, at an acceptable
utility cost: faithfulness remains competitive with the baseline, and latency is lower than the baseline
for single-attribute queries.}
\end{cvindent}
